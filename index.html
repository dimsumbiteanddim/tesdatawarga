Pencatatan Terstruktur Per KK & Warga

Input data cukup dilakukan 1 kali pada formulir Warga. Sistem akan memvalidasi No KK, secara otomatis mendaftarkan ke sheet DATA KK, menetapkan Kepala Keluarga, menghitung anggota, dan menggabungkan daftar nama secara otomatis.

Formulir Pendataan Warga (KTP & KK)

Lengkapi formulir di bawah ini. Pengelompokan KK akan terproses otomatis berdasarkan Nomor KK.

Validasi NIK & KK

1. Data Wilayah Domisili (Banjarmasin)

2. Dokumen Kependudukan (KK & NIK)

Wajib 16 digit angka sesuai dokumen Kartu Keluarga.

Primary key individu warga (unik).

3. Biodata Warga

Format Sesuai Sheet: DATA KK
No_KK Nama_Kepala_Keluarga Kelurahan ID_RT Alamat Jumlah_Anggota Daftar_Warga Aksi
Format Sesuai Sheet: DATA WARGA
No_Kartu_Keluarga Kelurahan RT NIK Nama_Lengkap Gender Agama Status_Keluarga Tgl_Lahir Alamat_Warga Pekerjaan Aksi

Script Google Apps Script (GAS) Backend

Salin kode di bawah ini lalu tempelkan ke **Apps Script** di Google Sheets Anda. Kode ini disesuaikan presisi dengan header kolom sheet DATA KK dan DATA WARGA. 

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Membuka Sheet Sesuai Nama Persis
    var sheetWarga = ss.getSheetByName("DATA WARGA");
    var sheetKK = ss.getSheetByName("DATA KK");
    
    if (!sheetWarga || !sheetKK) {
      return responseJSON({ status: "error", message: "Sheet 'DATA WARGA' atau 'DATA KK' tidak ditemukan!" });
    }

    var data = JSON.parse(e.postData.contents);

    // 1. TAMBAH BARIS KE SHEET 'DATA WARGA'
    // Kolom: No_Kartu_Keluarga | Kelurahan | RT | NIK | Nama_Lengkap | Gender | Agama | Status_Keluarga | Tgl_Lahir | Alamat_Warga | Pekerjaan
    sheetWarga.appendRow([
      "'" + data.no_kk,
      data.kelurahan,
      data.rt,
      "'" + data.nik,
      data.nama_lengkap,
      data.gender,
      data.agama,
      data.status_keluarga,
      data.tgl_lahir,
      data.alamat,
      data.pekerjaan
    ]);

    // 2. LOGIK KELOMPOK KK OTOMATIS DI SHEET 'DATA KK'
    // Kolom: No_KK | Nama_Kepala_Keluarga | Kelurahan | ID_RT | Alamat | Jumlah_Anggota | Daftar_Warga
    var dataKKValues = sheetKK.getDataRange().getValues();
    var kkFoundIndex = -1;

    for (var i = 1; i < dataKKValues.length; i++) {
      var existingNoKK = String(dataKKValues[i][0]).replace("'", "").trim();
      if (existingNoKK === String(data.no_kk).trim()) {
        kkFoundIndex = i + 1; // Baris spreadsheet (1-based index)
        break;
      }
    }

    if (kkFoundIndex !== -1) {
      // BILA NO KK SUDAH ADA (UPDATE ANGGOTA & DAFTAR WARGA)
      var currentKepala = sheetKK.getRange(kkFoundIndex, 2).getValue();
      var currentCount = parseInt(sheetKK.getRange(kkFoundIndex, 6).getValue()) || 0;
      var currentDaftar = String(sheetKK.getRange(kkFoundIndex, 7).getValue() || "");

      // Update Kepala Keluarga jika statusnya Kepala Keluarga
      if (data.status_keluarga === "Kepala Keluarga" || !currentKepala) {
        sheetKK.getRange(kkFoundIndex, 2).setValue(data.nama_lengkap);
      }

      // Update Jumlah Anggota +1
      sheetKK.getRange(kkFoundIndex, 6).setValue(currentCount + 1);

      // Append nama warga ke Daftar_Warga (koma pisah)
      var updatedDaftar = currentDaftar ? currentDaftar + ", " + data.nama_lengkap : data.nama_lengkap;
      sheetKK.getRange(kkFoundIndex, 7).setValue(updatedDaftar);

    } else {
      // BILA NO KK BELUM ADA (BUAT BARIS BARU DI SHEET DATA KK)
      var kepalaNama = (data.status_keluarga === "Kepala Keluarga") ? data.nama_lengkap : "";
      
      sheetKK.appendRow([
        "'" + data.no_kk,
        kepalaNama,
        data.kelurahan,
        data.rt,
        data.alamat,
        1,
        data.nama_lengkap
      ]);
    }

    return responseJSON({ status: "success", message: "Data Warga & Otomatisasi KK Berhasil Disimpan!" });

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

Langkah-Langkah Menghubungkan ke Google Sheets & GitHub Pages

  1. Buka file Spreadsheet Data Base Kependudukan Anda. Pastikan sheet bernama DATA KK dan DATA WARGA.
  2. Klik menu Ekstensi > Apps Script. Hapus semua kode lalu Salin Kode GAS di atas.
  3. Klik Deploy > Deployment Baru di kanan atas Apps Script.
  4. Pilih jenis Aplikasi Web (Web App). Atur Execute as ke Saya (Me), dan Who has access ke Siapa Saja (Anyone).
  5. Klik Deploy, izinkan akses akun Google Anda, lalu Salin URL Web App yang dihasilkan.
  6. Kembali ke aplikasi web ini, klik tombol Atur Web App URL di sidebar kiri, lalu tempelkan URL Web App tersebut.
